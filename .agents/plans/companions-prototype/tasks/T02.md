export type Appearance = 'skull' | 'wolf';
export type MissionKind = 'app' | 'research' | 'monitoring' | 'custom';

  | { status: 'paused'; resumeState: ResumeState }

  | { status: 'completed'; deliverableId: string; completedAt: string };

export type EventKind = 'created' | 'progress' | 'paused' | 'resumed' | 'goal_edited'

export interface ProgressEvent { id: string; companionId: string; missionId: string;
  kind: EventKind; summary: string; createdAt: string }
export interface Deliverable { id: string; companionId: string; missionId: string; title: string;
  kind: 'app' | 'report' | 'update' | 'note'; body: string; createdAt: string }

  missions: Mission[]; messages: Message[]; events: ProgressEvent[]; deliverables: Deliverable[] }
export interface Runtime { now(): string; id(): string }
export type ActionResult<T = undefined> = { ok: true; value: T }
  | { ok: false; error: { code: 'invalid_input' | 'not_found' | 'invalid_transition' | 'storage_error'; message: string } };

6. Enforce transitions centrally. Active and waiting-permission missions can pause; paused retains the runnable/permission state. Resume restores that state, including the same unresolved gate. Approval requires the current gate ID and returns active; decline stores the unresolved gate inside paused. Awaiting review is non-running and displays **Paused · Ready for review**; it offers confirm/corrections and rejects generic resume/advance. Confirm sets completed. Corrections from review or completed reopen the same mission as active, append the user request as a mission message/event, retain all history/deliverables and explicitly reset demoCursor=0 and progress=0 for a new correction cycle. Goal edits append an old/new-goal summary, preserve ID/history/state/gates and are rejected on completed until explicit corrections reopen it. Sending chat never changes the objective or approves/completes anything. Completed rejects simulation advancement. A partial deliverable leaves the mission active; only a final review step enters awaiting review. No generic unchecked `setStatus` action.

```ts
export type LoadResult = { kind: 'loaded'; snapshot: Snapshot } | { kind: 'empty' }
  | { kind: 'invalid' | 'unsupported_version' | 'unavailable'; message: string };
export type StorageResult = { ok: true } | { ok: false; message: string };
export interface Repository {
  load(): Promise<LoadResult>;
  save(snapshot: Snapshot): Promise<StorageResult>;
  reset(): Promise<StorageResult>;
}
export type StorageState = { mode: 'ready' }
  | { mode: 'memory' | 'blocked'; reason: 'unavailable' | 'save_failed' | 'invalid' | 'unsupported_version'; message: string };
```
